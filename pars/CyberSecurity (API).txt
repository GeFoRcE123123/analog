def cybersecurity_help():
    #выгрузка с сайта cybersecurity
    url = 'https://www.cybersecurity-help.cz/vdb/list.php'
    options = webdriver.ChromeOptions()
    options.add_argument('headless')
    options.add_argument('window-size=1920x1080')
    options.add_argument("disable-gpu")
    driver = webdriver.Chrome('chromedriver', chrome_options=options)
    driver.get(url)
    page = driver.page_source
    soup = BeautifulSoup(page,'html.parser')
    links = []
    for quote in soup.find_all('span',{'class':'cvp_title'}):
        lines = quote.find('a', href=re.compile("/vdb/"))
        links.append('https://www.cybersecurity-help.cz' + str(lines.attrs['href']))


    Description_list = []
    links_list = []
    cve_list = []
    CVSS_list = []

    for x in links:
        r = requests.get(x)
        soup = BeautifulSoup(r.text, 'lxml')
        try:
            for quote in soup.find_all('div',{'class':'cve'}):
                CVE = quote.find('p').findNext('p').findNext('p').find(text=re.compile('CVE-2'))
                a = (CVE.strip())

                if 'CVE' in a:
                    Description = quote.find('b',text=re.compile('Description')).findNext('p').findNext('p')
                    decs_dop = soup.find('div',{'class':'x_title'}).find('h2')
                    CVSS = quote.find('p').findNext('p').find('a',text=re.compile('CVSS'))

                    Description_list.append((str(decs_dop.text) + ' ' + str(Description.text)).replace('\r','').replace('\n','').replace('\t',''))
                    CVSS_list.append(float(re.sub(r"[^0-9.]", "", str(CVSS.text[0:3]))))
                    links_list.append(x)
                    cve_list.append(a)

        except:
            pass


    print(Description_list)
    print(links_list)
    print(cve_list)
    print(CVSS_list)
