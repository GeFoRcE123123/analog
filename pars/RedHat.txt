#Парсинг с REDHAT, links - ссылки, identifiers - идентификаторы ,etc - примечание ,lst_CVSSS - вектор (число FLOAT)
def red_hat():
    url = 'https://access.redhat.com/security/security-updates/#/cve?q=&p=1&sort=cve_publicDate%20desc&rows=100&documentKind=Cve'
    options = webdriver.ChromeOptions()
    options.add_argument('headless')
    options.add_argument('window-size=1920x1080')
    options.add_argument("disable-gpu")
    driver = webdriver.Chrome('chromedriver', chrome_options=options)
    driver.get(url)
    page = driver.page_source
    soup = BeautifulSoup(page,'html.parser')

    etc = []
    for quote in soup.find_all('span',{'ng-bind-html':'result.cve_details'}):
        lines = quote.text
        etc.append(lines)

    lst_CVSSS__ = []
    for quote in soup.find_all('td', {'class':'td-impact'}):
        lines = quote.text
        lst_CVSSS__.append(lines)
    lst_CVSSS_ = []
    for x in lst_CVSSS__:
        x = x.replace('Impact','').replace('\n','').replace('Moderate','5').replace('Low','3').replace('Important','9').replace('Critical','10')
        lst_CVSSS_.append(x)
    lst_CVSSS = []
    for x in lst_CVSSS_:
        if x != '':
            lst_CVSSS.append(float(x))
        else:
            lst_CVSSS.append(float(0))

    identifiers = []
    for quote in soup.find_all('span',{'class':'cell-content'}):
        lines = quote.find('a')
        if lines != None:
            identifiers.append(lines.text)

    links = []
    for quote in soup.find_all('tr', {'pagination-id':'CVE'}): 
        lines = quote.find('a')
        links.append(lines.attrs['href'])

    source_in = 'redhat'
