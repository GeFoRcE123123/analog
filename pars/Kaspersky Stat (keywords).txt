def kaspersky_stat():
    #парсинг CVE с kaspersky_stat и добавление
    urls = ['https://statistics.securelist.com/ru/vulnerability-scan/week','https://statistics.securelist.com/ru/vulnerability-scan/day','https://statistics.securelist.com/ru/vulnerability-scan/month']

    options = webdriver.ChromeOptions()
    options.add_argument('headless')
    options.add_argument('window-size=1920x1080')
    options.add_argument("disable-gpu")
    driver = webdriver.Chrome('chromedriver', chrome_options=options)

    identifiers = []
    lst_CVSSS = []

        driver.get(url)
        page = driver.page_source

            links.append(quote.attrs['href'])

            identifiers.append(((a[a.find("CVE") : ])[ : (a[a.find("CVE") : ]).find(".")]))

            a = soup.find('div', {'id':'GeneratedTable'}).find('table').find('tbody').findChildren('tr')[3].find('td')
            etc.append((a.text).replace('\n',''))
