#Парсинг с fortiguard, links - ссылки, identifiers - идентификаторы ,etc - примечание ,lst_CVSSS - вектор (число FLOAT)
def fortiguard():
    url = 'https://www.fortiguard.com/zeroday'
    r = requests.get(url)
    soup = BeautifulSoup(r.text, 'lxml')

    links = []
    for quote in soup.find_all('div',{'class':'title'}):
        lines = quote.find('a')
        links.append(lines.attrs['href'])
    links = ["https://www.fortiguard.com" + suit for suit in links]


    identifiers = []
    for quote in soup.find_all('div',{'class':'line'}):
        lines = quote.find('a',href=re.compile("http://cve.mitre.org/")).text
        identifiers.append(lines)

    etc = []
    for quote in soup.find_all('div',{'class':'title'}):
        lines = quote.a.find_next_sibling('a').text
        etc.append(lines)

    w = []
    for quote in soup.find_all('div',{'class':'line'}):
        lines = quote.find('a')
        w.append(lines.attrs['href'])

    w_ = []
    for x in w:
        w_.append(re.sub('\D', '', x))

    source_in = 'fortiguard'

    lst_CVSSS = [float(x)*2 for x in w_]

    etc = [re.sub(r"[^а-яА-ЯёЁa-zA-Z0-9-./#!?@: ]", "", str(x)) for x in etc]

    for link_in,CVE_in,etc_in,CVSS_in in zip(links,identifiers,etc,lst_CVSSS):
