def cvedetails():
    #парсинг для ключевых слов с cvedetails и их обновление
    url = 'https://www.cvedetails.com/top-50-product-cvssscore-distribution.php'

    options = webdriver.ChromeOptions()
    options.add_argument('headless')
    options.add_argument('window-size=1920x1080')
    options.add_argument("disable-gpu")
    driver = webdriver.Chrome('chromedriver', chrome_options=options)
    driver.get(url)
    page = driver.page_source
    soup = BeautifulSoup(page,'html.parser')

    name = []
    price = []
    i = 2
    while True:
        try:   
            lines1 = soup.find('table', {'class':'grid'}).find('tbody').findChildren('tr')[i].findChildren('td')[1].findChildren('a')[0]
            lines2 = soup.find('table', {'class':'grid'}).find('tbody').findChildren('tr')[i].findChildren('td')[14]
            name.append(lines1.text)
            price.append(float(lines2.text)*10)
            i = i + 1
        except:
            break
    for name_,price_ in zip(name,price):
        while True:
            sqlite_select_query = """SELECT * FROM dictionary WHERE word = %s"""

            record = cursor.fetchone()
            if record is None:
                a = """INSERT INTO dictionary
                    (word, price)
                    VALUES
                    (%s,%s);"""

                break
