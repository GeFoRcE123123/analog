#Парсинг с postgresql,  links - ссылки, identifiers - идентификаторы ,etc - примечание ,lst_CVSSS - вектор (число FLOAT)
def postgresql():
    url = 'https://www.postgresql.org/support/security/'
    r = requests.get(url)
    soup = BeautifulSoup(r.text, 'lxml')

    etc_ = []
    for quote in soup.find_all('td'):
        if quote is not None:
            lines = quote.next_sibling.next_sibling
        etc_.append(lines)
    etc_ = list(filter(None.__ne__, etc_))
    records_cleaned = [str(x) for x in etc_]
    records_cleaned = list(records_cleaned)
    www = [s for s in records_cleaned if "more details" in s]
    etc = []
    left = '<td>'
    right = '<br/><br/><a'
    for x in www:
        x_ = x[x.index(left)+len(left):x.index(right)]
        etc.append(x_)


    identifiers = []
    for quote in soup.find_all('nobr'):
        lines = quote.find('a').text
        identifiers.append(lines)

    links = []
    for quote in soup.find_all('nobr'):
        lines = quote.find('a')
        links.append(lines.attrs['href'])
    links = ["https://www.postgresql.org" + suit for suit in links]

    lst_CVSSS = []
    for quote in soup.find_all("a",href=re.compile("nvd")): 
        lines = quote.text
        lst_CVSSS.append(float(lines))

    source_in = 'PostgreSQL'
