def cisco():
    #выгрузка с сайта cisco
    url = 'https://tools.cisco.com/security/center/psirtrss20/CiscoSecurityAdvisory.xml'

    options = webdriver.ChromeOptions()
    options.add_argument('headless')
    options.add_argument('window-size=1920x1080')
    options.add_argument("disable-gpu")
    driver = webdriver.Chrome('chromedriver', chrome_options=options)

    driver.get(url)
    page = driver.page_source
    soup = BeautifulSoup(page,'xml')

    links = []
    identifiers = []
    identifiers_ = []
    lst_CVSSS = []
    etc = []
    etc_ = []
    links_ = []

    for quote in soup.find_all('description')[1:]:
        etc_.append(quote.text)

    for quote in soup.find_all('link')[2:]:
        links_.append(quote.text)

    for quote in soup.find_all('item'):
        w = quote.text
        a = re.findall(r"CVE-2\w+-\w+",w)
        a = list(set(a))
        identifiers_.append(a)

    for y,x,w in zip(links_,identifiers_,etc_):
        for z in x:
          identifiers.append(z)
          links.append(y)
          etc.append(w)


    try:
        for x in links:
            driver.get(x)
            page = driver.page_source
            soup = BeautifulSoup(page,'html.parser')
            try:
                a = soup.find('div', {'class':'udheadercol1'}).find('div', {'class':'flexcol'}).find('div', {'class':'ud-CVSSScore divPaddingTen'}).find('div', {'class':'divLabelContent'}).find('div').find('a',text=re.compile("."))
                lst_CVSSS.append(a.text)
            except:
                a = '0'
                lst_CVSSS.append(a)
    except:
            Box = QMessageBox()
            Box.setText("Error")
            Box.setInformativeText("Запустите парсинг еще раз, ошибка отклика!")
            Box.setStandardButtons(QMessageBox.Close)
            Box.setDefaultButton(QMessageBox.Close)
            ret = Box.exec_()


    lst_CVSSS = [re.sub(r"[^0-9-.]", "", str(x)) for x in lst_CVSSS]
    lst_CVSSS = [float(item) for item in lst_CVSSS]       

    source_in = 'cisco'
    for link_in,CVE_in,etc_in,CVSS_in in zip(links,identifiers,etc,lst_CVSSS):
