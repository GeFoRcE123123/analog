    mkdir -p temp_bdu && cd temp_bdu
    curl -L --insecure -o vulxml.zip \
      "https://bdu.fstec.ru/files/documents/vulxml.zip"
    unzip vulxml.zip && cd ..
    
    # Импортировать
    python services/parsers/bdu_importer.py \
      --xml-file temp_bdu/export/vulxml.xml
